// src/utils/classifyWine.ts

export type WineTypeKey =
  | "red"
  | "white"
  | "rose"
  | "sparkling"
  | "orange"
  | "fortified"
  | "dessert"
  | "nonAlcoholic";

export interface WineInput {
  name: string;
  producer?: string;
  vintage?: string | number;
  region?: string;
  grapes?: string[] | string;
  style?: string;
  notes?: string;
}

export interface WineClassification {
  typePrimary: WineTypeKey;
  typeSecondary: WineTypeKey[];
  confidence: number;
  reasons: string[];
}

/* ---------------------------------------------------- */
/* ------------------- CONSTANTS ---------------------- */
/* ---------------------------------------------------- */

const SPARKLING_KEYWORDS = [
  "champagne",
  "cava",
  "prosecco",
  "franciacorta",
  "sekt",
  "cremant",
  "spumante",
  "pet-nat",
  "pet nat"
];

const ROSE_KEYWORDS = ["rosé", "rose"];
const ORANGE_KEYWORDS = ["orange", "skin-contact", "skin contact", "ramato"];
const FORTIFIED_KEYWORDS = [
  "port",
  "porto",
  "sherry",
  "madeira",
  "marsala",
  "vin doux naturel",
  "banyuls",
  "rivesaltes"
];

const DESSERT_KEYWORDS = [
  "late harvest",
  "icewine",
  "sauternes",
  "tokaji",
  "beerenauslese",
  "trockenbeerenauslese",
  "passito",
  "vin santo",
  "demi-sec",
  "doux"
];

const NON_ALCOHOLIC_KEYWORDS = [
  "non-alcoholic",
  "dealcoholized",
  "0.0"
];

const RED_GRAPES = [
  "cabernet sauvignon",
  "merlot",
  "pinot noir",
  "syrah",
  "shiraz",
  "grenache",
  "tempranillo",
  "sangiovese",
  "nebbiolo",
  "malbec",
  "zinfandel",
  "barbera",
  "mourvedre",
  "gamay",
  "carmenere"
];

const WHITE_GRAPES = [
  "chardonnay",
  "sauvignon blanc",
  "riesling",
  "pinot grigio",
  "pinot gris",
  "chenin blanc",
  "albarino",
  "gruner veltliner",
  "viognier",

  "fiano"
];

/* ---------------------------------------------------- */
/* ----------------- HELPERS -------------------------- */
/* ---------------------------------------------------- */

function normalize(text?: string): string {
  if (!text) return "";
  return text
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/\s+/g, " ")
    .trim();
}

function containsAny(haystack: string, needles: string[]): string | null {
  for (const needle of needles) {
    if (haystack.includes(needle)) return needle;
  }
  return null;
}

/* ---------------------------------------------------- */
/* ----------------- MAIN FUNCTION -------------------- */
/* ---------------------------------------------------- */

export function classifyWine(wine: WineInput): WineClassification {
  const reasons: string[] = [];

  const searchable = normalize(
    [
      wine.name,
      wine.region,
      wine.style,
      wine.notes,
      Array.isArray(wine.grapes) ? wine.grapes.join(" ") : wine.grapes
    ]
      .filter(Boolean)
      .join(" ")
  );

  // 1️⃣ Non-alcoholic (highest priority)
  const nonAlcoholicMatch = containsAny(searchable, NON_ALCOHOLIC_KEYWORDS);
  if (nonAlcoholicMatch) {
    reasons.push(`keyword:${nonAlcoholicMatch}`);
    return {
      typePrimary: "nonAlcoholic",
      typeSecondary: [],
      confidence: 1,
      reasons
    };
  }

  // 2️⃣ Sparkling
  const sparklingMatch = containsAny(searchable, SPARKLING_KEYWORDS);
  if (sparklingMatch) {
    reasons.push(`keyword:${sparklingMatch}`);
    return {
      typePrimary: "sparkling",
      typeSecondary: [],
      confidence: 0.95,
      reasons
    };
  }

  // 3️⃣ Rosé
  const roseMatch = containsAny(searchable, ROSE_KEYWORDS);
  if (roseMatch) {
    reasons.push(`keyword:${roseMatch}`);
    return {
      typePrimary: "rose",
      typeSecondary: [],
      confidence: 0.95,
      reasons
    };
  }

  // 4️⃣ Orange
  const orangeMatch = containsAny(searchable, ORANGE_KEYWORDS);
  if (orangeMatch) {
    reasons.push(`keyword:${orangeMatch}`);
    return {
      typePrimary: "orange",
      typeSecondary: [],
      confidence: 0.95,
      reasons
    };
  }

  // 5️⃣ Fortified
  const fortifiedMatch = containsAny(searchable, FORTIFIED_KEYWORDS);
  if (fortifiedMatch) {
    reasons.push(`keyword:${fortifiedMatch}`);
    return {
      typePrimary: "fortified",
      typeSecondary: [],
      confidence: 0.95,
      reasons
    };
  }

  // 6️⃣ Dessert
  const dessertMatch = containsAny(searchable, DESSERT_KEYWORDS);
  if (dessertMatch) {
    reasons.push(`keyword:${dessertMatch}`);
    return {
      typePrimary: "dessert",
      typeSecondary: [],
      confidence: 0.9,
      reasons
    };
  }

  // 7️⃣ Grape-based inference
  const grapeString = normalize(
    Array.isArray(wine.grapes)
      ? wine.grapes.join(" ")
      : wine.grapes
  );

  if (grapeString) {
    const redMatch = containsAny(grapeString, RED_GRAPES);
    if (redMatch) {
      reasons.push(`grape:${redMatch}`);
      return {
        typePrimary: "red",
        typeSecondary: [],
        confidence: 0.85,
        reasons
      };
    }

    const whiteMatch = containsAny(grapeString, WHITE_GRAPES);
    if (whiteMatch) {
      reasons.push(`grape:${whiteMatch}`);
      return {
        typePrimary: "white",
        typeSecondary: [],
        confidence: 0.85,
        reasons
      };
    }
  }

  // 8️⃣ Language fallback
  if (searchable.includes("rouge") || searchable.includes("rosso")) {
    reasons.push("language:red");
    return {
      typePrimary: "red",
      typeSecondary: [],
      confidence: 0.7,
      reasons
    };
  }

  if (searchable.includes("blanc") || searchable.includes("bianco")) {
    reasons.push("language:white");
    return {
      typePrimary: "white",
      typeSecondary: [],
      confidence: 0.7,
      reasons
    };
  }

  // 9️⃣ Final safe fallback
  return {
    typePrimary: "white",
    typeSecondary: [],
    confidence: 0.4,
    reasons: ["fallback:default_white"]
  };
}
